const store: Record<string, string> = {};

// Simulate RecentModelSelectionManager.getStorage
function getStorageKey(profile: string) {
  return `recent-model-selection:${profile}`;
}

function getStorage(profile: string): string[] {
  try {
    const key = getStorageKey(profile);
    const stored = store[key] ?? null;
    return stored ? JSON.parse(stored) : [];
  } catch {
    return [];
  }
}

function setStorage(profile: string, models: string[]): void {
  try {
    const key = getStorageKey(profile);
    store[key] = JSON.stringify(models);
  } catch {
    // Ignore
  }
}

function addRecentSelection(profile: string, providerSlug: string, modelSlug: string): boolean {
  const key = `${providerSlug}:${modelSlug}`;
  const all = getStorage(profile);
  console.log('Storage key:', key);
  console.log('Current storage before:', JSON.stringify(all));
  if (!all.includes(key)) {
    all.push(key);
    setStorage(profile, all);
    console.log('Added, storage after:', JSON.stringify(all));
    return true;
  }
  console.log('Duplicate detected');
  return false;
}
