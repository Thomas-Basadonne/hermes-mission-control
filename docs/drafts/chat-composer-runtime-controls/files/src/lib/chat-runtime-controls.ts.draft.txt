import type { ChatModelIdentity } from './chat-protocol';

const CODEX_LEGACY = ['none', 'low', 'medium', 'high', 'xhigh'];
const CODEX_WIDE = [...CODEX_LEGACY, 'max'];
const GRADED = ['none', 'low', 'medium', 'high', 'max'];
const VALID_EFFORTS = new Set(['none', 'minimal', 'low', 'medium', 'high', 'xhigh', 'max']);

// Conservative route vocabulary, matching agent/reasoning_effort.py. Unknown
// models get no control: a configured effort alone does not prove capability.
export function reasoningOptionsForModel(identity: ChatModelIdentity | null | undefined): string[] {
  if (!identity?.model || !identity.provider) return [];
  const provider = identity.provider.toLowerCase();
  const model = identity.model.toLowerCase().split('/').at(-1) || '';
  if (provider === 'openai-codex') {
    if (/^gpt-6-astra(?:-900k)?$/.test(model)) return CODEX_WIDE.filter((value) => value !== 'none');
    const wide = model.includes('gpt-5.6') || /^gpt-6-(sol|luna)/.test(model) || /^gpt-daybreak-blue-latest/.test(model);
    return [...(wide ? CODEX_WIDE : CODEX_LEGACY)];
  }
  if (/^glm-5\.2/.test(model)) return ['none', 'high', 'max'];
  if (/^glm-5\.3|^deepseek-v4/.test(model)) return [...GRADED];
  if (/^kimi-k3|^k3(?:$|-)/.test(model)) return ['none', 'low', 'high', 'max'];
  if (/^kimi-k2/.test(model)) return ['none', 'low', 'medium', 'high'];
  if (/^grok-4\.6/.test(model)) return ['low', 'medium', 'high', 'xhigh'];
  if (/^grok-[34]/.test(model)) return ['low', 'medium', 'high'];
  if (/^claude-(sonnet|opus)-[45]|^qwen3(?:\.|:|-)|^minimax-m[23]|^nemotron-3/.test(model)) return ['none', 'low', 'medium', 'high'];
  if (provider === 'openai' && /^gpt-[56]|^o[134](?:-|$)/.test(model)) return [...CODEX_LEGACY];
  return [];
}

type RuntimeRequest = (method: string, params: Record<string, unknown>) => Promise<unknown>;

export async function setChatReasoning(value: string, sessionId: string, request: RuntimeRequest): Promise<string> {
  if (!sessionId) throw new Error('Session required for reasoning selection.');
  if (!VALID_EFFORTS.has(value)) throw new Error('Invalid reasoning effort.');
  await request('config.set', { key: 'reasoning', value, scope: 'session', session_id: sessionId });
  const verified = await request('config.get', { key: 'reasoning', session_id: sessionId });
  const actual = verified && typeof verified === 'object' && 'value' in verified ? String(verified.value) : '';
  if (actual !== value) throw new Error('Reasoning change was not confirmed by the gateway.');
  return actual;
}
