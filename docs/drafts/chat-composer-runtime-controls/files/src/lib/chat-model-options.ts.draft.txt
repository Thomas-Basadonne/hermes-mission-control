export type ChatModelProviderOption = {
  slug: string;
  name: string;
  models: string[];
  total_models: number;
  is_current?: boolean;
  authenticated?: boolean;
  warning?: string;
};

function asRecord(value: unknown): Record<string, unknown> | null {
  return value && typeof value === 'object' && !Array.isArray(value) ? value as Record<string, unknown> : null;
}

export function normalizeProvider(rawProvider: unknown): ChatModelProviderOption | null {
  const raw = asRecord(rawProvider);
  if (!raw || typeof raw.slug !== 'string' || !raw.slug.trim()) return null;

  const models = Array.isArray(raw.models)
    ? raw.models.filter((model): model is string => typeof model === 'string' && model.trim().length > 0)
    : [];

  return {
    slug: raw.slug.trim(),
    name: typeof raw.name === 'string' && raw.name.trim() ? raw.name.trim() : raw.slug.trim(),
    models,
    total_models: typeof raw.total_models === 'number' ? raw.total_models : models.length,
    is_current: raw.is_current === true,
    authenticated: raw.authenticated !== false,
    warning: typeof raw.warning === 'string' ? raw.warning : undefined,
  };
}

export class RecentModelSelectionManager {
  private readonly profile: string;

  constructor(profile: string = 'default') {
    this.profile = profile;
  }

  private getStorageKey(): string {
    return `recent-model-selection:${this.profile}`;
  }

  private getStorage(): string[] {
    try {
      const stored = typeof window !== 'undefined' && window.localStorage ? window.localStorage.getItem(this.getStorageKey()) : null;
      return stored ? JSON.parse(stored) : [];
    } catch {
      return [];
    }
  }

  private setStorage(models: string[]): void {
    try {
      if (typeof window !== 'undefined' && window.localStorage) {
        window.localStorage.setItem(this.getStorageKey(), JSON.stringify(models));
      }
    } catch {
      // Ignore storage failures
    }
  }

  getRecentModels(providerSlug: string): string[] {
    const all = this.getStorage();
    return all.filter(key => key.startsWith(`${providerSlug}:`))
      .map(key => key.substring(providerSlug.length + 1));
  }

  addRecentSelection(providerSlug: string, modelSlug: string): boolean {
    const key = `${providerSlug}:${modelSlug}`;
    const all = this.getStorage();
    if (!all.includes(key)) {
      all.push(key);
      this.setStorage(all);
      return true;
    }
    return false;
  }

  getAllRecentSelections(): Array<{ provider: string; model: string }> {
    const all = this.getStorage();
    return all.map(key => {
      const idx = key.indexOf(':');
      if (idx <= 0) return { provider: key, model: '' };
      return { provider: key.substring(0, idx), model: key.substring(idx + 1) };
    }).filter(item => item.provider.length > 0 && item.model.length > 0);
  }

  clearRecentForProvider(providerSlug: string): void {
    const all = this.getStorage();
    const filtered = all.filter(key => !key.startsWith(`${providerSlug}:`));
    this.setStorage(filtered);
  }

  hasRecentSelection(providerSlug: string, modelSlug: string): boolean {
    return this.getRecentModels(providerSlug).includes(modelSlug);
  }
}

export function filterModels(query: string, models: string[]): string[] {
  console.log('filterModels called with query:', query, 'models:', models);
  if (!query.trim()) return models;
  const lowerQuery = query.toLowerCase();
  const result = models.filter(model => {
    const lowerModel = model.toLowerCase();
    const includes = lowerModel.includes(lowerQuery);
    console.log('Checking model:', model, 'lower:', lowerModel, 'includes:', includes);
    return includes;
  });
  console.log('filterModels result:', result);
  return result;
}

export function isCurrentModel(model: string, currentModel: string | null): boolean {
  return Boolean(currentModel && model === currentModel);
}

export function isProviderAvailable(provider: ChatModelProviderOption): boolean {
  return Boolean(provider.authenticated && provider.models.length > 0);
}

export function isModelSelected(model: string, selectedModel: string): boolean {
  return model === selectedModel;
}

export function providerMatches(query: string, provider: ChatModelProviderOption): boolean {
  if (!query.trim()) return true;
  const lowerQuery = query.toLowerCase();
  return provider.slug.toLowerCase().includes(lowerQuery) || provider.name.toLowerCase().includes(lowerQuery);
}

export function modelMatches(query: string, model: string): boolean {
  if (!query.trim()) return true;
  return model.toLowerCase().includes(query.toLowerCase());
}

export function isProviderSelected(selectedProvider: string, provider: ChatModelProviderOption): boolean {
  return selectedProvider === provider.slug;
}

export function isValidModel(model: string): boolean {
  return typeof model === 'string' && model.trim().length > 0;
}
