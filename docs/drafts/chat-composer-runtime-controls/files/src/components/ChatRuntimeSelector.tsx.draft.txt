import { useEffect, useRef, useState } from 'react';
import { ChatModelPicker } from './ChatModelPicker';
import type { ChatModelIdentity, ChatModelSwitchResult } from '../lib/chat-protocol';
import { reasoningOptionsForModel } from '../lib/chat-runtime-controls';

type Props = {
  identity: ChatModelIdentity | null;
  sessionId: string | null;
  profile?: string;
  request: <T,>(method: string, params?: Record<string, unknown>) => Promise<T>;
  modelOpen: boolean;
  refresh: boolean;
  onOpenModel: () => void;
  onCloseModel: () => void;
  onModelChange: (model: string, provider: string, confirm?: boolean) => Promise<ChatModelSwitchResult>;
  onReasoningChange: (value: string) => Promise<void>;
  onBusyChange: (busy: boolean) => void;
  running: boolean;
  disabled: boolean;
};

export function ChatRuntimeSelector({ identity, sessionId, profile, request, modelOpen, refresh, onOpenModel, onCloseModel, onModelChange, onReasoningChange, onBusyChange, running, disabled }: Props) {
  const [reasoningOpen, setReasoningOpen] = useState(false);
  const [busy, setBusy] = useState(false);
  const [error, setError] = useState<string | null>(null);
  const [pending, setPending] = useState<{ sessionId: string; value: string } | null>(null);
  const rootRef = useRef<HTMLDivElement>(null);
  const modelButtonRef = useRef<HTMLButtonElement>(null);
  const reasoningButtonRef = useRef<HTMLButtonElement>(null);
  const sessionRef = useRef(sessionId);
  sessionRef.current = sessionId;
  const options = reasoningOptionsForModel(identity);
  const queued = pending?.sessionId === sessionId ? pending : null;
  const effort = queued?.value || identity?.reasoningEffort || 'default';
  const close = () => { onCloseModel(); setReasoningOpen(false); };

  useEffect(() => {
    setPending(null);
    setError(null);
    setReasoningOpen(false);
    onCloseModel();
  }, [sessionId, onCloseModel]);

  useEffect(() => {
    if (!modelOpen) return;
    setReasoningOpen(false);
    setError(null);
  }, [modelOpen]);

  useEffect(() => {
    if (!modelOpen && !reasoningOpen) return;
    const dismiss = (event: PointerEvent) => {
      if (!rootRef.current?.contains(event.target as Node)) close();
    };
    const escape = (event: KeyboardEvent) => {
      if (event.key !== 'Escape') return;
      event.preventDefault();
      (modelOpen ? modelButtonRef : reasoningButtonRef).current?.focus();
      close();
    };
    document.addEventListener('pointerdown', dismiss);
    document.addEventListener('keydown', escape);
    return () => {
      document.removeEventListener('pointerdown', dismiss);
      document.removeEventListener('keydown', escape);
    };
  }, [modelOpen, reasoningOpen, onCloseModel]);

  useEffect(() => {
    if (running || disabled || busy || !queued) return;
    const targetSession = queued.sessionId;
    setBusy(true);
    onBusyChange(true);
    setPending(null);
    void onReasoningChange(queued.value).catch((err) => {
      if (sessionRef.current === targetSession) setError(err instanceof Error ? err.message : 'Could not change reasoning.');
    }).finally(() => { setBusy(false); onBusyChange(false); });
  }, [running, disabled, busy, queued, onReasoningChange, onBusyChange]);

  const chooseReasoning = async (value: string) => {
    if (running && sessionId) {
      setPending({ sessionId, value });
      setReasoningOpen(false);
      return;
    }
    const targetSession = sessionId;
    setBusy(true);
    onBusyChange(true);
    setError(null);
    try {
      await onReasoningChange(value);
      setReasoningOpen(false);
      reasoningButtonRef.current?.focus();
    } catch (err) {
      if (sessionRef.current === targetSession) setError(err instanceof Error ? err.message : 'Could not change reasoning.');
    } finally {
      setBusy(false);
      onBusyChange(false);
    }
  };

  return (
    <div className="chat-runtime-selector" ref={rootRef}>
      <button ref={modelButtonRef} className="chat-runtime-model" type="button" disabled={disabled || busy} aria-label="Choose provider and model" aria-haspopup="dialog" aria-expanded={modelOpen} title={identity ? `${identity.provider || 'Provider'} · ${identity.model}${running ? ' · Changes apply next turn' : ''}` : 'Choose provider and model'} onClick={() => { setError(null); setReasoningOpen(false); if (modelOpen) onCloseModel(); else onOpenModel(); }}>
        {identity?.provider ? <span className="chat-runtime-provider">{identity.provider}</span> : null}
        {identity?.provider ? <span className="chat-runtime-divider" aria-hidden> / </span> : null}
        <span className="chat-runtime-model-name">{identity?.model || 'Choose model'}</span>
      </button>
      {options.length ? (
        <button ref={reasoningButtonRef} className={`chat-runtime-reasoning ${queued ? 'is-pending' : ''}`} type="button" disabled={disabled || busy} aria-label={`Reasoning: ${effort}${queued ? ', next turn' : ''}`} aria-haspopup="dialog" aria-expanded={reasoningOpen} title={queued ? 'Queued for next turn' : 'Reasoning effort'} onClick={() => { onCloseModel(); setError(null); setReasoningOpen((open) => !open); }}>
          <span className="chat-runtime-reasoning-label">Reasoning </span><strong>{busy ? 'saving…' : effort}</strong>{queued ? <span className="chat-runtime-next"> · next</span> : null}
        </button>
      ) : null}
      {modelOpen ? (
        <div className="chat-runtime-popover" role="dialog" aria-label="Choose provider and model">
          {running ? <p className="chat-runtime-notice">Changes apply next turn.</p> : null}
          <ChatModelPicker request={request} sessionId={sessionId} profile={profile} currentModel={identity?.model} currentProvider={identity?.provider} initialRefresh={refresh} variant="popover" onClose={() => { onCloseModel(); modelButtonRef.current?.focus(); }} onSelect={onModelChange} />
        </div>
      ) : reasoningOpen ? (
        <div className="chat-runtime-popover chat-reasoning-picker" role="dialog" aria-label="Choose reasoning effort">
          <header><strong>Reasoning</strong><button type="button" onClick={() => { setReasoningOpen(false); reasoningButtonRef.current?.focus(); }}>Close</button></header>
          {running ? <p className="chat-runtime-notice">Changes apply next turn.</p> : null}
          <div className="chat-reasoning-options">
            {options.map((value) => <button key={value} type="button" aria-pressed={effort === value} disabled={busy} onClick={() => void chooseReasoning(value)}>{value === 'none' ? 'Off' : value}</button>)}
          </div>
          {error ? <p className="chat-runtime-error" role="alert">{error}</p> : null}
        </div>
      ) : null}
      {error && !reasoningOpen ? <p className="chat-runtime-error chat-runtime-inline-error" role="alert">{error}</p> : null}
    </div>
  );
}
