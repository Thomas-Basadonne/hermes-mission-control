import {
  normalizeProvider,
  RecentModelSelectionManager,
  filterModels,
  isCurrentModel,
  isProviderAvailable,
  isModelSelected,
  providerMatches,
  modelMatches,
  isProviderSelected
} from '../src/lib/chat-model-options.ts';

function assertEqual<T>(actual: T, expected: T) {
  if (actual !== expected) {
    throw new Error(`Expected ${JSON.stringify(expected)}, got ${JSON.stringify(actual)}`);
  }
}

function assertDeepEqual(actual: unknown, expected: unknown) {
  const actualJson = JSON.stringify(actual);
  const expectedJson = JSON.stringify(expected);
  if (actualJson !== expectedJson) {
    throw new Error(`Expected ${expectedJson}, got ${actualJson}`);
  }
}

// Mock localStorage for tests
const outboxStorage = new Map<string, string>();
const testWindow = {
  localStorage: {
    getItem: (key: string) => outboxStorage.get(key) ?? null,
    setItem: (key: string, value: string) => { outboxStorage.set(key, value); },
    removeItem: (key: string) => { outboxStorage.delete(key); }
  }
};
const globalWithWindow = globalThis as unknown as { window?: typeof testWindow };
const previousWindow = globalWithWindow.window;
globalWithWindow.window = testWindow;

console.log('Testing chat-model-options...');

const testProvider = {
  slug: 'ollama',
  name: 'Ollama',
  models: ['llama3', 'phi3', 'mistral'],
  total_models: 3,
  is_current: true,
  authenticated: true,
  warning: 'Rate limited'
};

const normalizedProvider = normalizeProvider(testProvider);
if (!normalizedProvider) throw new Error('Provider should normalize correctly');

assertEqual(normalizedProvider.slug, 'ollama');
assertEqual(normalizedProvider.name, 'Ollama');
assertDeepEqual(normalizedProvider.models, ['llama3', 'phi3', 'mistral']);
assertEqual(normalizedProvider.total_models, 3);
assertEqual(normalizedProvider.is_current, true);
assertEqual(normalizedProvider.authenticated, true);
assertEqual(normalizedProvider.warning, 'Rate limited');

const emptyProvider = {
  slug: 'empty',
  name: 'Empty',
  models: [],
  total_models: 0,
  is_current: false,
  authenticated: true
};

const normalizedEmpty = normalizeProvider(emptyProvider);
if (!normalizedEmpty) throw new Error('Empty provider should normalize correctly');
assertEqual(normalizedEmpty.models.length, 0);
assertEqual(normalizedEmpty.total_models, 0);

const invalidProvider = null;
assertEqual(normalizeProvider(invalidProvider), null);

const manager = new RecentModelSelectionManager('test-profile');
outboxStorage.clear();

const addedFirst = manager.addRecentSelection('provider-a', 'model-1');
assertEqual(addedFirst, true);

const addedDuplicate = manager.addRecentSelection('provider-a', 'model-1');
assertEqual(addedDuplicate, false);

const addedSecond = manager.addRecentSelection('provider-a', 'model-2');
assertEqual(addedSecond, true);

const recentModels = manager.getRecentModels('provider-a');
assertDeepEqual(recentModels, ['model-1', 'model-2']);

const allRecent = manager.getAllRecentSelections();
assertEqual(allRecent.length, 2);
assertDeepEqual(allRecent[0], { provider: 'provider-a', model: 'model-1' });
assertDeepEqual(allRecent[1], { provider: 'provider-a', model: 'model-2' });

manager.clearRecentForProvider('provider-a');
assertEqual(manager.getRecentModels('provider-a').length, 0);
assertEqual(manager.getAllRecentSelections().length, 0);

assertEqual(manager.hasRecentSelection('provider-a', 'model-1'), false);
assertEqual(manager.hasRecentSelection('provider-b', 'model-x'), false);

manager.addRecentSelection('provider-a', 'model-1');
assertEqual(manager.hasRecentSelection('provider-a', 'model-1'), true);
assertEqual(manager.hasRecentSelection('provider-a', 'model-2'), false);

const models = ['alpha', 'beta', 'gamma', 'delta'];
const filteredA = filterModels('a', models);
console.log('filteredA:', filteredA);
assertDeepEqual(filteredA, ['alpha', 'gamma']);
const filteredEmpty = filterModels('', models);
assertDeepEqual(filteredEmpty, models);
const filteredCase = filterModels('A', ['Alpha', 'BETA', 'gamma']);
assertDeepEqual(filteredCase, ['Alpha']);

assertEqual(isCurrentModel('model-1', 'model-1'), true);
assertEqual(isCurrentModel('model-1', 'model-2'), false);
assertEqual(isCurrentModel('model-1', null), false);

const availableProvider = {
  slug: 'available',
  name: 'Available',
  models: ['model1'],
  total_models: 1,
  is_current: false,
  authenticated: true
};
assertEqual(isProviderAvailable(availableProvider), true);

const unavailableProviderModels = {
  slug: 'unavailable-models',
  name: 'Unavailable Models',
  models: [],
  total_models: 0,
  is_current: false,
  authenticated: true
};
assertEqual(isProviderAvailable(unavailableProviderModels), false);

const unavailableProviderAuth = {
  slug: 'unavailable-auth',
  name: 'Unavailable Auth',
  models: ['model1'],
  total_models: 1,
  is_current: false,
  authenticated: false
};
assertEqual(isProviderAvailable(unavailableProviderAuth), false);

assertEqual(isModelSelected('model-1', 'model-1'), true);
assertEqual(isModelSelected('model-1', 'model-2'), false);

assertEqual(providerMatches('test', { slug: 'test-provider', name: 'Test Provider', models: [], total_models: 0, is_current: false, authenticated: true }), true);
assertEqual(providerMatches('TEST', { slug: 'test-provider', name: 'Test Provider', models: [], total_models: 0, is_current: false, authenticated: true }), true);
assertEqual(providerMatches('other', { slug: 'test-provider', name: 'Test Provider', models: [], total_models: 0, is_current: false, authenticated: true }), false);

assertEqual(modelMatches('test', 'test-model'), true);
assertEqual(modelMatches('MODEL', 'test-model'), true);
assertEqual(modelMatches('xyz', 'test-model'), false);

assertEqual(isProviderSelected('test-provider', { slug: 'test-provider', name: 'Test Provider', models: [], total_models: 0, is_current: false, authenticated: true }), true);
assertEqual(isProviderSelected('other-provider', { slug: 'test-provider', name: 'Test Provider', models: [], total_models: 0, is_current: false, authenticated: true }), false);

if (previousWindow) globalWithWindow.window = previousWindow;
else delete globalWithWindow.window;

console.log('All tests passed!');
