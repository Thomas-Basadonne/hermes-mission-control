import assert from 'node:assert/strict';
import { reasoningOptionsForModel, setChatReasoning } from '../src/lib/chat-runtime-controls.ts';

assert.deepEqual(reasoningOptionsForModel({ provider: 'openai-codex', model: 'gpt-5.6-luna' }), ['none', 'low', 'medium', 'high', 'xhigh', 'max']);
assert.deepEqual(reasoningOptionsForModel({ provider: 'openai-codex', model: 'gpt-5.5' }), ['none', 'low', 'medium', 'high', 'xhigh']);
assert.deepEqual(reasoningOptionsForModel({ provider: 'openai-codex', model: 'gpt-6-astra' }), ['low', 'medium', 'high', 'xhigh', 'max']);
assert.deepEqual(reasoningOptionsForModel({ provider: 'ollama-cloud', model: 'gemma4:31b' }), []);
assert.deepEqual(reasoningOptionsForModel({ provider: 'custom', model: 'unknown-model' }), []);
assert.deepEqual(reasoningOptionsForModel(null), []);
assert.deepEqual(reasoningOptionsForModel({ provider: 'ollama-cloud', model: 'glm-5.2:cloud' }), ['none', 'high', 'max']);
assert.deepEqual(reasoningOptionsForModel({ provider: 'deepseek', model: 'deepseek-v4-flash' }), ['none', 'low', 'medium', 'high', 'max']);

const calls: unknown[] = [];
const result = await setChatReasoning('high', 'session-a', async (method, params) => {
  calls.push({ method, params });
  return { value: 'high' };
});
assert.equal(result, 'high');
assert.deepEqual(calls, [{ method: 'config.set', params: { key: 'reasoning', value: 'high', scope: 'session', session_id: 'session-a' } }, { method: 'config.get', params: { key: 'reasoning', session_id: 'session-a' } }]);
await assert.rejects(() => setChatReasoning('ultra', 'session-a', async () => ({})), /Invalid reasoning/);
await assert.rejects(() => setChatReasoning('high', '', async () => ({})), /Session/);
await assert.rejects(() => setChatReasoning('high', 'session-a', async (method) => method === 'config.set' ? { value: 'high' } : { value: 'low' }), /not confirmed/);
console.log('chat runtime controls tests passed');
